-- 1. TABLA PARA MEDICIONES DE TEMPERATURA, HUMEDAD Y ESTADO DE LUZ
CREATE TABLE IF NOT EXISTS telemetria (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    tipo VARCHAR(20) NOT NULL, -- Guardará: 'temperatura', 'humedad' o 'luz'
    valor NUMERIC NOT NULL     -- Guardará el valor numérico leído
);

-- 2. TABLA PARA EL REGISTRO HISTÓRICO DE ALERTAS
CREATE TABLE IF NOT EXISTS alertas (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    tipo_alerta VARCHAR(30) NOT NULL, -- Guardará: 'FLAMA', 'TERREMOTO', 'PROX' o 'SONIDO'
    descripcion TEXT,
    estado VARCHAR(20) DEFAULT 'ACTIVA'
);

-- 3. TABLA PARA MANTENER EL ÚLTIMO ESTADO DEL FOCO (1 sola fila activa)
CREATE TABLE IF NOT EXISTS estado_sistema (
    id INT PRIMARY KEY DEFAULT 1,
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    nivel_luz INT DEFAULT 0,
    CONSTRAINT single_row CHECK (id = 1)
);

-- Insertar la fila inicial fija para la luz
INSERT INTO estado_sistema (id, nivel_luz) 
VALUES (1, 0) 
ON CONFLICT (id) DO NOTHING;

ALTER TABLE alertas DISABLE ROW LEVEL SECURITY;
ALTER TABLE estado_sistema DISABLE ROW LEVEL SECURITY;
